    GalleryControl.cs (attached to ground plane)

    using System.Collections;

    using System.Collections.Generic;

    using UnityEngine;

    public class GalleryControl : MonoBehaviour

    {

    //PSEUDO CODE: Declare(Create) a public variable for an array, Type: GameObject

    //FUNCTION: Can Populate(Initialize) in Inspector Box and hold multiple Game Objects....

    public GameObject[] MecaBrickItems;

    //PSEUDO CODE: Declare(Create) and Initialize(give value) a public variable for counting

    //FUNCTION: Starts at 0, used to access array items in sequence....

    public int current = 0;

    void Update()

    {

    //PSEUDO CODE: Check for Key input (RightArrow)

    if (Input.GetKeyDown(KeyCode.RightArrow))

    {

    //PSEUDO CODE: if there are still game objects in array, increase counter by one

    if (current <= MecaBrickItems.Length - 1)

    {

    current++;

    //FUNCTION: To move up the array

    }

    //PSEUDO CODE:at the end of array, reset counter to go back to start of array

    if (current == MecaBrickItems.Length)

    {

    current = 0;

    //FUNCTION: to cycle the array

    //TIP: Don't forget Arrays start at 0!

    }

    }

    //PSEUDO CODE: Check for Key input (LeftArrow)

    if (Input.GetKeyDown(KeyCode.LeftArrow))

    {

    //PSEUDO CODE: if there are still game objects in array, decrease counter by one

    if (current >= 0)

    {

    current--;

    //FUNCTION: To move down the array

    }

    //PSEUDO CODE: at the beginning of array, reset counter to go back to end of array

    if (current == -1)

    {

    current = MecaBrickItems.Length - 1;

    //FUNCTION: to cycle the array

    }

    }

    Debug.Log("Current" + current);

    //PSEUDO CODE: Print to Console log the value of the counter

    //FUNCTION: To check the keys work accurately

    //PSEUDO CODE: Using counter, loop through Array, turn the Game Objects on and off

    //FUNCTION: This displays and hide the items as controlled by the keys

    //TIP: Check if each array item matches the counter with the debug -Current- value

    // Use a Component of GameObjects to display/hide as needed

    // Look in Unity reference manual

    for (int i = 0; i<MecaBrickItems.Length; i++)

    {

    if (i != current)

    {

    MecaBrickItems[i].SetActive(false);

    }

    else

    {

    MecaBrickItems[current].SetActive(true);

    }

    }

    }

    }
